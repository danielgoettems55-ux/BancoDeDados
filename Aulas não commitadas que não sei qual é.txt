select tc.tx_nome,
		max(tp.vl_preco), 
		min(tp.vl_preco)
from tb_categoria tc 
	inner join tb_produto tp on tc.id_categoria = tp.cd_categoria 
	group by tc.tx_nome;
	 
select tx_nome, 
		sum(tpedido.nr_quantidade  * tp.vl_preco) as soma
from tb_produto tp 
	inner join tb_pedido tpedido on tp.id_produto = tpedido.cd_produto 
	group by tx_nome
	order by soma desc;

	add constraint cliente_pk foreign key(cd_cliente) references tb_cliente(id_cliente);
