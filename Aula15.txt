select usuario.tx_nome,
		item.id_item_pedido,
		produto.tx_nome,
		item.nr_quantidade,
		vl_unitario
		from tb_pedidos tp 
		inner join tb_usuarios usuario on tp.cd_usuario = usuario.id_usuario 
		inner join tb_itens_pedido item  on item.cd_pedido  = tp.id_pedido  
		inner join tb_produtos produto on produto.id_produto  = item.cd_produto 
		where usuario.id_usuario =1234;

create index
	idx_cd_produto
		on tb_itens_pedido(cd_produto);
		 
create index
	idx_cd_pedido
	on tb_itens_pedido(cd_pedido);
