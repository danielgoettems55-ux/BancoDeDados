		tp.nr_estoque 
		from tb_produto tp 
		inner join tb_categoria tv on tp.cd_categoria = tv.id_categoria 
			where tp.fl_ativo is true;

with catego as (
select cat.tx_nome,
		count(id_categoria) as total
		from tb_produto tp 
		inner join tb_categoria cat on tp.cd_categoria  = cat.id_categoria
		group by cat.tx_nome
)
select * from catego where total > 2
