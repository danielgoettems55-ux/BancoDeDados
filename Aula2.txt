Recriação tabela cliente:

 create table tb_cliente(
	id_cliente serial primary key not null,
	tx_nome varchar(80) not null,
	nr_telefone numeric(18,0) not null,
	nr_idade numeric(18,0) not null,
	tx_cpf varchar(11) not null,
	tx_email varchar(50) not null,
	tx_senha varchar(30) not null,
	dt_data timestamp default current_date
 );

-----------------------------------------------------------------------------------

Exibir todos os itens de uma tabela:

select * from tb_cliente

Criação da tabela calculadora:

create table tb_calculadora(
	id_calculadora serial primary key not null,
	vl_valor numeric(18,2) not null,
	tx_operacao varchar(1) not null
);
