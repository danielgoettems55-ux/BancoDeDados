drop function fn_nome
create function fn_nome(valor numeric)
returns varchar
language plpgsql
as $$
begin	
	if(valor > 18) then
		return 'maior de idade';
	elsif(valor = 18)then
		return 'exatamento 18 anos';
	else
		return 'menor de idade';
	end if;
end
$$;

		fn_nome(50)
from tb_atividade;

drop function fn_idade;
create function fn_idade(dt_nascimento date)
returns varchar
language plpgsql
as 
$$
declare
	idade integer;
begin
	idade := extract(year from age(dt_nascimento) );
	if(idade >18 ) then
		return 'maior de idade';
		
	elsif(idade=18) then
		return 'tem 18 anos';
		
	else
		return 'menor de idade';
	end if; 
end
$$
