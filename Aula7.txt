--1

select tb_produto.tx_nome, tb_categoria.tx_nome , vl_preco, nr_estoque from tb_produto 
	inner join tb_categoria on 	
		tb_produto.cd_categoria = tb_categoria.id_categoria
			where fl_ativo = true and vl_preco between 500 and 1000;

--2

select tx_nome, tx_cpf, tx_email from tb_cliente 
	where UPPER(tx_estado) in (upper('SC'), upper('PR'))
	order by tx_nome asc;

--3

select tb_cliente.tx_nome as nomeUsuario , tb_cliente.nr_idade as idadeUsuario 
from tb_cliente
	where tx_email = null or nr_idade > 50;

--4

select 
tb_produto.tx_nome as nomeProduto,
tb_fornecedor.tx_nome as nomeFornecedor,
tb_produto.vl_preco as valorProduto,
tb_categoria.tx_descricao as descricaoCategoria

from tb_produto 
	
	inner join tb_fornecedor on
	 tb_produto.cd_fornecedor = tb_fornecedor.id_fornecedor
	
	 inner join tb_categoria on
	  tb_produto.cd_categoria = tb_categoria.id_categoria
	  
	  where tb_produto.fl_ativo = true
	  order by tb_produto.vl_preco desc;
	  
--5

select 
	tx_nome as nomeCliente,
	tx_cidade as cidadeCliente,
	count(venda.id_venda) as quantidadeVendas
	
from tb_cliente cliente
	left join tb_venda venda on
		cliente.id_cliente = venda.cd_cliente 
			group by tx_nome, tx_cidade
				order by quantidadeVendas desc;

--6

select 
	categoria.tx_nome
	
from tb_categoria categoria
	left join tb_produto produto on 
		categoria.id_categoria = produto.cd_categoria 
			where id_produto is null
			order by tx_nome asc;

--7

select 
	tx_nome as nomeCliente,
	sum(itenVenda.vl_unitario * itenVenda.nr_quantidade)
	
	
from tb_cliente cliente
	left join tb_venda venda on
		cliente.id_cliente = venda.cd_cliente 
			inner join tb_item_venda itenVenda on 
				venda.id_venda = itenVenda.cd_venda 
				
			group by tx_nome, tx_cidade;
