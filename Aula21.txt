select
	tbc.tx_nome as tx_cliente,
	tbpl.tx_nome as tx_plano,
	count(tbco.id_cobranca) as nr_cobranca , 
	max(tbco.vl_cobranca) as vl_maior_cobranca,
	sum(tbco.vl_cobranca) as vl_cobrado,
	sum(case when id_pagamento is not null then tbco.vl_cobranca else 0 end) as vl_pago,
	sum(case when id_pagamento is null then tbco.vl_cobranca else 0 end) as vl_em_aberto,
	max(tbpa.dt_pagamento) as dt_primeiro_pagamento,
	min(tbpa.dt_pagamento) as dt_ultimo_pagamento
from
	tb_clientes tbc
inner join tb_planos tbpl on
	tbc.cd_plano = tbpl.id_plano
inner join tb_cobrancas tbco on
	tbco.cd_cliente = tbc.id_cliente
left join tb_pagamentos tbpa on
	tbco.id_cobranca = tbpa.cd_cobranca
group by
	tbc.tx_nome ,
	tbpl.tx_nome 
order by vl_em_aberto desc, tbc.tx_nome asc
