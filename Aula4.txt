Atividade para criar tokens de privilégio

create table tb_usuario(
	id_usuario serial,
	cd_privilegio integer not null,
	tx_email varchar(255) not null,
	tx_senha varchar (255) not null,

	constraint pk_tb_usuario primary key (id_usuario),
	constraint fk_tb_privilegio_id_prilegio foreign key (cd_privilegio) references tb_privilegio(id_privilegio)
)

create table tb_privilegio(
	id_privilegio serial not null,
	tx_nome varchar(255) not null,

	constraint pk_tb_privilegio primary key (id_privilegio)
)
